```lua server.lua
local S = UgCore.Schema

UgCore.Callback.Register('Withdraw', {
    schema = S.Define({ S.Integer({ min = 1, max = 100000 }) }),
    rate = { max = 3, per = 5000 },
    requirePlayer = true,
}, function(source, amount)
    local ok, balance, errorCode = UgCore.Accounts.Remove(source, 'bank', amount, 'ATM withdrawal')

    return { ok = ok, balance = balance, error = errorCode }
end)
```

```lua client.lua
local ok, result, errorCode = UgCore.Callback.Await('my-bank:Withdraw', 500)

if not ok then
    -- errorCode is one of: rate_limited, invalid_args, not_loaded, timeout, ...
    return
end

print(('New balance: %d'):format(result.balance))
```
